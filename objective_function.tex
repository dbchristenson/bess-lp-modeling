\section*{Objective Function}

The objective of the sizing and operation problem is to minimize the risk-adjusted
annualized life-cycle cost (ALCC). This formulation is simplified from
Ljungblom~(2025, Eq.~3.1): PV, wind, and SMR generation terms are dropped
(grid~+~BESS only), grid export is removed, and a revenue discount factor
captures auction uncertainty for larger DSU bids.

\begin{equation}
\boxed{
\begin{aligned}
\min \; Z \;=\; & F_{\text{sub}} \\[6pt]
& + \underbrace{
    \text{AF}_{\text{BESS}}
    \left(c_P^{\text{cap}}\, P_{\text{BESS}}
        + c_E^{\text{cap}}\, E_{\text{BESS}}\right)
    + c_P^{\text{opex}}\, P_{\text{BESS}}
    + c_E^{\text{opex}}\, E_{\text{BESS}}
}_{\text{BESS CapEx + O\&M}} \\[6pt]
& + \underbrace{
    \sum_{t \in \mathcal{T}}
    \left(\lambda_t + \pi_t\right) i_t
}_{\text{Grid import cost}} \\[6pt]
& - \underbrace{
    \left(1 - \beta\,\frac{P_{\text{BESS}}}{P_{\max}}\right)
    R_{\text{DR}}
}_{\text{Risk-adjusted DR revenue}}
\end{aligned}
}
\tag{1}
\end{equation}

\noindent where the demand response revenue $R_{\text{DR}}$ is defined as:

\begin{equation}
R_{\text{DR}} \;=\;
    r^{\text{cap}}\, C_{\text{enr}}
    \cdot \mathbb{1}\!\left(C_{\text{enr}} \geq C_{\min}\right)
\tag{2}
\end{equation}

\subsection*{Subject to}

\begin{alignat}{3}
& \text{Energy balance:}
    &\quad & i_t + d_t = D + c_t
    && \quad \forall\, t \in \mathcal{T}
    \tag{3} \\[4pt]
& \text{State of charge:}
    && s_t = s_{t-1} + \eta_{\text{c}}\, c_t - d_t
    && \quad \forall\, t \in \mathcal{T},\; t > 0
    \tag{4} \\[4pt]
& \text{Initial SoC:}
    && s_0 = S_0 + \eta_{\text{c}}\, c_0 - d_0
    &&
    \tag{5} \\[4pt]
& \text{Wrap-around:}
    && s_{|\mathcal{T}|-1} \geq S_0
    &&
    \tag{6} \\[4pt]
& \text{Charge bound:}
    && 0 \leq c_t \leq P_{\text{BESS}}
    && \quad \forall\, t \in \mathcal{T}
    \tag{7} \\[4pt]
& \text{Discharge bound:}
    && 0 \leq d_t \leq P_{\text{BESS}}
    && \quad \forall\, t \in \mathcal{T}
    \tag{8} \\[4pt]
& \text{SoC bound:}
    && 0 \leq s_t \leq E_{\text{BESS}}
    && \quad \forall\, t \in \mathcal{T}
    \tag{9} \\[4pt]
& \text{Import bound:}
    && 0 \leq i_t \leq C_{\text{grid}}
    && \quad \forall\, t \in \mathcal{T}
    \tag{10}
\end{alignat}
